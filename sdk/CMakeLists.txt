cmake_minimum_required(VERSION 3.22)
project(SpoolProviderContract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
add_executable(provider-contract-runner provider-contract-runner.cpp)
target_link_libraries(provider-contract-runner PRIVATE Qt6::Core Qt6::Qml)
install(TARGETS provider-contract-runner RUNTIME DESTINATION bin)
