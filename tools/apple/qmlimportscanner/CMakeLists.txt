cmake_minimum_required(VERSION 3.22)
project(SpoolAppleQmlImportScanner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml QmlCompilerPrivate)
if(NOT EXISTS "${SPOOL_QTDECLARATIVE_SOURCE}/tools/qmlimportscanner/main.cpp")
    message(FATAL_ERROR "SPOOL_QTDECLARATIVE_SOURCE must name the verified pinned Qt Declarative sources")
endif()
add_executable(spool-qmlimportscanner "${SPOOL_QTDECLARATIVE_SOURCE}/tools/qmlimportscanner/main.cpp")
target_compile_definitions(spool-qmlimportscanner PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
target_link_libraries(spool-qmlimportscanner PRIVATE Qt6::QmlCompilerPrivate Qt6::QmlPrivate)
